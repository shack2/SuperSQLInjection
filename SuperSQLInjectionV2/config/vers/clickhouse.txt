当前数据库：currentDatabase()
数据库版本：version()
当前用户：currentUser()
服务器名：hostName()
服务器运行秒数：uptime()
数据库列表：(select name from system.databases order by name limit 1)
全部用户：(select arrayStringConcat(groupArray(name),',') from system.users)
当前用户权限：(select arrayStringConcat(groupArray(access_type),',') from system.grants where user_name=currentUser())
全部角色：(select arrayStringConcat(groupArray(name),',') from system.roles)
