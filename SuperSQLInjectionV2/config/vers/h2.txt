数据库版本：h2version()
当前模式：current_schema()
当前用户：current_user
数据库路径：(select setting_value from information_schema.settings where setting_name='info.BASE_DIR' union all select setting_value from information_schema.settings where setting_name='property.user.dir' limit 1)
Java版本：(select setting_value from information_schema.settings where setting_name='java.version' union all select setting_value from information_schema.settings where setting_name='property.java.runtime.version' limit 1)
管理员权限：(select cast(is_admin as varchar) from information_schema.users where user_name=current_user)
